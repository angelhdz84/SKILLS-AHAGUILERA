> **For agentic workers:** REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement 6 improvements (3 Lite + 3 Full) + transversal Web Workers for IA Jutia.

**Architecture:** Each improvement is a self-contained change to existing templates. Lite and Full share the same module structure (`core/ia.js` + `module.html` + `module.js`) but are in separate template directories. The Lite worker is a new file; the Full worker already exists and will be extended with OCR support.

**Tech Stack:** Alpine.js, Dexie.js, FlexSearch, DaisyUI, Tesseract.js (Full only), Transformers.js (Full only)

**Design doc:** `docs/superpowers/specs/2026-06-29-ia-jutia-v02-design.md`

```javascript
    _highlight(texto, query) {
      if (!texto || !query) return texto || '';
      const words = query.trim().split(/\s+/).filter(w => w.length > 1);
      let result = String(texto).slice(0, 500);
      words.forEach(w => {
        const re = new RegExp(`(${w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
        result = result.replace(re, '<mark class="bg-warning/30 text-warning-content px-0.5 rounded">$1</mark>');
      });
      return result;
    },

    _groupResults(results) {
      const groups = {};
      results.forEach(r => {
        const key = r.tabla || 'otros';
        if (!groups[key]) groups[key] = { tabla: key, items: [], count: 0 };
        groups[key].items.push(r);
        groups[key].count++;
      });
      return Object.values(groups).sort((a, b) => b.count - a.count);
    },

    _contextSnippet(texto, query, maxLen = 100) {
      if (!texto || !query) return texto?.slice(0, maxLen) || '';
      const idx = texto.toLowerCase().indexOf(query.toLowerCase());
      if (idx === -1) return texto.slice(0, maxLen);
      const start = Math.max(0, idx - 40);
      const end = Math.min(texto.length, idx + query.length + 60);
      let snippet = (start > 0 ? '...' : '') + texto.slice(start, end) + (end < texto.length ? '...' : '');
      return snippet;
    },
```

```javascript
    async sugerir() {
      if (!this.query || this.query.length < 2 || !window.ia?._flex) {
        this.sugerencias = [];
        this.mostrarSugerencias = false;
        return;
      }
      const results = window.ia._flex.search(this.query, { limit: 8, suggest: true, enrich: true });
      const flat = [];
      for (const res of results) {
        for (const item of res.result || []) {
          if (item.doc && !flat.find(f => f.id === item.doc.id)) {
            flat.push(item.doc);
          }
        }
      }
      this.sugerencias = flat.slice(0, 8);
      this.mostrarSugerencias = this.sugerencias.length > 0;
      this.selectedSug = -1;

    seleccionarSugerencia(idx) {
      const sug = this.sugerencias[idx];
      if (sug) {
        this.query = sug.nombre || sug.tabla || '';
        this.mostrarSugerencias = false;
        this.buscar();
      }
    },

    onSugKeydown(e) {
      if (!this.mostrarSugerencias) return;
      if (e.key === 'ArrowDown') {
        e.preventDefault();
        this.selectedSug = Math.min(this.selectedSug + 1, this.sugerencias.length - 1);
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        this.selectedSug = Math.max(this.selectedSug - 1, 0);
      } else if (e.key === 'Enter' && this.selectedSug >= 0) {
        e.preventDefault();
        this.seleccionarSugerencia(this.selectedSug);
      } else if (e.key === 'Escape') {
        this.mostrarSugerencias = false;
        this.selectedSug = -1;
      }
```

```javascript
// core/ia-worker.js — Lite: FlexSearch en Web Worker
// FlexSearch inyectado inline vía blob URL desde ia.js
// Mensajes: search, registerTable, indexRecord, removeRecord

let flex = null;

self.addEventListener('message', (e) => {
  const { type, id, data } = e.data;
  try {
    if (type === 'init') {
      const FlexSearch = self.FlexSearch;
      if (!FlexSearch) {
        self.postMessage({ type: 'error', id, error: 'FlexSearch no disponible en Worker' });
        return;
      }
      flex = new FlexSearch.Document({
        document: { id: 'id', index: ['nombre', 'descripcion', 'notas', 'texto'], store: ['nombre', 'tipo', 'tabla', 'texto', 'descripcion'] },
        tokenize: 'forward',
        cache: true
      });
      self.postMessage({ type: 'init-done', id });
      return;
    }

    if (type === 'add') {
      if (!flex) { self.postMessage({ type: 'error', id, error: 'Worker no iniciado' }); return; }
      const { docs } = data;
      docs.forEach(d => flex.add(d));
      self.postMessage({ type: 'add-done', id, count: docs.length });
      return;
    }

    if (type === 'search') {
      if (!flex) { self.postMessage({ type: 'error', id, error: 'Worker no iniciado' }); return; }
      const { query, limit } = data;
      const results = flex.search(query, { limit: limit || 20, enrich: true, suggest: true });
      const flat = [];
      for (const res of results) {
        for (const item of res.result || []) {
          if (item.doc) flat.push(item.doc);
        }
      }
      self.postMessage({ type: 'search-result', id, data: flat });
      return;
    }

    if (type === 'remove') {
      if (!flex) return;
      flex.remove(data.id);
      self.postMessage({ type: 'remove-done', id });
      return;
    }
  } catch (err) {
    self.postMessage({ type: 'error', id, error: err.message });
  }
});
```

```javascript
    _initWorker() {
      if (typeof Worker === 'undefined') { this._workerReady = false; return; }
      try {
        // Inyectar FlexSearch inline en Worker via blob URL
        const flexSource = typeof FlexSearch === 'function' ? FlexSearch.toString() : 'function(){}';
        const blob = new Blob([`
          self.FlexSearch = (${flexSource});
          ${this._workerSource()}
        `], { type: 'application/javascript' });
        this._worker = new Worker(URL.createObjectURL(blob));
        this._worker.onmessage = (e) => {
          const { type, id, data, error } = e.data;
          if (type === 'init-done') {
            this._workerReady = true;
            this._flushQueue();
            return;
          }
          const cb = this._workerCallbacks[id];
          if (cb) {
            delete this._workerCallbacks[id];
            if (type === 'error') cb.reject(new Error(error));
            else cb.resolve(data);
          }
        };
        this._worker.postMessage({ type: 'init', id: 'init' });
      } catch (e) {
        console.warn('ia-jutia: Worker no disponible, usando hilo principal');
        this._workerReady = false;
      }
    },

    _workerSource() {
      return self.addEventListener ? self.addEventListener.toString().split('{')[1].split('}')[0] : '';
    },

    _workerPost(type, data) {
      return new Promise((resolve, reject) => {
        const id = ++this._workerReqId;
        this._workerCallbacks[id] = { resolve, reject };
        if (this._workerReady) {
          this._worker.postMessage({ type, id, data });
        } else {
          this._workerQueue.push({ type, id, data });
        }
      });
    },

    _flushQueue() {
      this._workerQueue.forEach(msg => this._worker.postMessage(msg));
      this._workerQueue = [];
    },
```

```javascript
    async searchHybrid(query, opts = {}) {
      // 1. FlexSearch results (keyword)
      const flexResults = await this.search(query, opts);
      if (!this._embedPipeline || !this._modelosCargados) {
        return flexResults; // fallback a solo keyword
      }

      try {
        // 2. Embedding de la query
        const queryEmb = await this._embedPipeline(query, { pooling: 'mean', normalize: true });
        const queryVec = Array.from(queryEmb.data);

        // 3. Buscar chunks con embeddings en Dexie
        const BATCH = 50;
        const MAX_BATCHES = 3;
        const totalChunks = await window.db._ia_chunks.count();
        let semanticResults = [];

        for (let offset = 0; offset < Math.min(totalChunks, BATCH * MAX_BATCHES); offset += BATCH) {
          const batch = await window.db._ia_chunks.offset(offset).limit(BATCH).toArray();
          for (const chunk of batch) {
            if (chunk.embedding) {
              const score = this._cosineSimilarity(queryVec, chunk.embedding);
              if (score > 0.15) {
                semanticResults.push({ id: `chunk-${chunk.docId}`, nombre: chunk.texto.slice(0, 80), texto: chunk.texto.slice(0, 200), tipo: 'documento', tabla: '_ia_docs', _semanticScore: score });
              }
            }
          }
        }
        semanticResults.sort((a, b) => b._semanticScore - a._semanticScore);
        semanticResults = semanticResults.slice(0, 10);

        // 4. Combinar: 0.6 flex + 0.4 semantic
        const combined = {};
        flexResults.forEach(r => {
          combined[r.id] = { ...r, _combinedScore: 0.6 };
        });
        semanticResults.forEach(r => {
          if (combined[r.id]) {
            combined[r.id]._combinedScore = 0.6 + 0.4 * r._semanticScore;
          } else {
            combined[r.id] = { ...r, _combinedScore: 0.4 * r._semanticScore };
          }
        });
        const finalResults = Object.values(combined).sort((a, b) => b._combinedScore - a._combinedScore).slice(0, 20);

        if (typeof Alpine !== 'undefined') {
          Alpine.store('ia').results = finalResults;
          Alpine.store('ia').searching = false;
        }
        return finalResults;
      } catch (e) {
        console.warn('⚠️ ia-jutia: Error en busqueda hibrida, usando solo FlexSearch', e.message);
        return flexResults;
      }
```

```javascript
    _cosineSimilarity(a, b) {
      if (!a || !b || a.length !== b.length) return 0;
      let dot = 0, normA = 0, normB = 0;
      for (let i = 0; i < a.length; i++) {
        dot += a[i] * b[i];
        normA += a[i] * a[i];
        normB += b[i] * b[i];
      }
      const denom = Math.sqrt(normA) * Math.sqrt(normB);
      return denom ? dot / denom : 0;
    },
```
